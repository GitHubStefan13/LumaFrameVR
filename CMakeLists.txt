cmake_minimum_required(VERSION 3.24)
project(LumaFrameVR VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(LUMAFRAMEVR_LINK_WRAP "--wrap=" CACHE STRING "Linker wrap option for the active linker")
include(cmake/I18n.cmake)
include(cmake/FolderBrowser.cmake)
include(cmake/ProductionInterface.cmake)
include(cmake/ControllerMath.cmake)
include(cmake/VideoPresentation.cmake)
option(LUMAFRAMEVR_BUILD_VULKAN "Build Vulkan resource ownership and shader validation" ON)
if(LUMAFRAMEVR_BUILD_VULKAN)
  include(cmake/VulkanValidation.cmake)
endif()
option(LUMAFRAMEVR_BUILD_MEDIA "Build the native multimedia subsystem" ON)
if(LUMAFRAMEVR_BUILD_MEDIA)
  include(cmake/NativeMedia.cmake)
  include(cmake/Images.cmake)
endif()
add_library(lumaframevr_source cpp/source/media_source.cpp)
target_include_directories(lumaframevr_source PUBLIC cpp/source)
target_compile_options(lumaframevr_source PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(cmake/ApplicationOptions.cmake)
if(LUMAFRAMEVR_BUILD_VULKAN AND LUMAFRAMEVR_BUILD_MEDIA)
  include(cmake/OpenXrApplication.cmake)
  include(cmake/Probe.cmake)
  include(cmake/AdditionalComponents.cmake)
  include(cmake/ProfileCommands.cmake)
  include(cmake/ApplicationAssembly.cmake)
  include(cmake/ApplicationRuntime.cmake)
  endif()
include(cmake/Tests.cmake)
include(cmake/Packaging.cmake)
